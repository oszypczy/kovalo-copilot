    reference_usd: float            # wszystkie tokeny po cenie cost.reference_model (PINNED, cennik kursu)
    unpriced_models: list[str] = [] # modele z tokenami > 0 bez ceny (posortowane, bez powtórzeń)
    our_prices_checked_on: date     # cost.our_prices_checked_on (UI może pokazać w podpowiedzi)

def group_by_model(calls: Iterable[Mapping[str, Any]]) -> list[ModelUsage]:

    Wywołanie bez `usage` liczy się do `calls` z zerowymi tokenami. reasoning_tokens: suma albo None,
    gdy żadne wywołanie modelu ich nie podało."""

def price_for_model(model: str, prices: Mapping[str, TokenPrice]) -> TokenPrice | None:
    """Usuwa prefiks "models/"; dokładne trafienie; inaczej najdłuższy klucz K, dla którego nazwa zaczyna się
    od K + "-" (np. wersja z sufiksem daty); inaczej None. Bez zgadywania po rodzinie modelu."""

def price_usage(usage: Sequence[ModelUsage], cost: CostConfig) -> TurnCost:
    """Pozycje z model None i zerowymi tokenami pomija; model None z tokenami > 0 → "nieznany" w unpriced_models.
    Kwoty bez zaokrąglania (zaokrągla widok). Używa pricing.tokens_cost, tej samej funkcji co raporty E3–E5."""

### 6.4 UI (`ui/client.py`, `ui/render.py`, `ui/app.py`)

```python
class ModelUsageView(_View): model: str | None = None; calls: int; input_tokens: int; output_tokens: int
class TurnCostView(_View): ours_usd: float | None = None; reference_usd: float; unpriced_models: list[str] = []
class UsageView(_View):
    calls: int; input_tokens: int; output_tokens: int; reasoning_tokens: int | None = None
    models: list[str] = []; by_model: list[ModelUsageView] = []; cost: TurnCostView | None = None
class TurnView(...):  # F5 + usage: UsageView | None = None

@dataclass(frozen=True)
class UsageTotal:
    turns: int = 0; calls: int = 0; input_tokens: int = 0; output_tokens: int = 0
    ours_usd: float = 0.0; reference_usd: float = 0.0
    partial: bool = False            # któraś tura bez ceny (ours_usd None albo cost None)

def add_usage(total: UsageTotal, usage: UsageView | None) -> UsageTotal   # None → total bez zmian
def render_usage(usage: UsageView | None, total: UsageTotal) -> str | None # None, gdy usage None
```

Treść linii (Markdown, kursywa, ostatnia linia ostatniej wiadomości tury, po linku do ścieżki dowodowej; przy pauzie po pytaniu o akceptację):

```
_Tura: 5 wywołań (gemini-3.5-flash-lite), 9120 tokenów wejścia, 612 wyjścia. Koszt: 0 USD w darmowym tierze;
po cenach płatnych 0,43 centa (Gemini) / 0,96 centa (cennik kursu). Rozmowa: 2 tury, 0,71 / 1,52 centa._
```

- Liczby całkowite przez `kovalo.text_pl.number` (spacja tysięcy od pięciu cyfr: „9120”, „62 092”; P1 z §14). Centy z dwoma miejscami; wartość dodatnia poniżej 0,005 centa → „< 0,01 centa”. Odmiana „wywołanie / wywołania / wywołań” (1; 2–4 poza 12–14; reszta). Modele: lista z `by_model` bez powtórzeń, przecinkami (fallback widoczny).
- `ours_usd None` → „po cenach płatnych: brak ceny dla <model>” i `partial` w sumie („Rozmowa: … (część tur bez ceny)”). `cost None` (API bez cennika) → linia bez części o koszcie.
- Suma rozmowy: `cl.user_session["usage_total"]`, aktualizowana przez `add_usage` po każdym `final`; nowa rozmowa = nowa suma (rozmowa i wątek żyją tyle co sesja Chainlita, U1).
- Ani „0 USD w darmowym tierze”, ani nazwa cennika kursu nie są wpisane jako nazwy modeli: nazwę modelu UI bierze z `usage.by_model`, a cennik kursu opisuje słowami.

## 7. Zamknięcie przerwanej tury (R1-8 z F5, U13 = B z F5)

Wątek jest „przerwany”, gdy ostatni checkpoint ma `snapshot.next != ()` i nie ma aktywnej pauzy (`_first_pending(snapshot.interrupts) is None`). Powstaje, gdy proces `api` zginie w trakcie węzła (po `stop_grace_period`) albo gdy węzeł zgłosi wyjątek (`internal_error`). Dziś każda kolejna wiadomość dostaje 409 `thread_stuck`.

```python
# workflow/replies.py
INTERRUPTED_TURN_REPLY = (
    "Poprzednia odpowiedź nie została dokończona (usługa została przerwana). "
    "Jeśli nadal jej potrzebujesz, zadaj pytanie ponownie."
)

# workflow/service.py
def _close_interrupted_turn(self, config: RunnableConfig, snapshot: StateSnapshot) -> StateSnapshot:
    """graph.update_state(config, {"status": "error", "error": "turn_interrupted", "calls": [],
    "messages": [AIMessage(INTERRUPTED_TURN_REPLY)]}, as_node="reply_other") — reply_other ma krawędź do END,
    więc nowy checkpoint ma next == (). Zwraca nowy snapshot. Log WARNING: thread_id, przerwane kroki (snapshot.next).
    Wyjątek update_state (np. InvalidUpdateError) → ThreadStuckError jak w F5; błąd połączenia → CheckpointerUnavailableError."""
```

- **Kiedy (U7 = A):** leniwie, pod blokadą wątku, w `start_message` i `start_resume`, zaraz po odczycie stanu. Blokada wyklucza turę w toku w tym procesie, a `api` ma jeden proces Uvicorna (§11.1), więc `next != ()` bez pauzy znaczy na pewno „nikt tej tury nie dokończy”. Bez skanu wątków przy starcie (nie ma rejestru wątków, ADR-0010).
- **Co z turą (U7 = A):** zamykamy, nie dokańczamy. Ponowne uruchomienie węzła agenta wydałoby limit i dałoby odpowiedź na pytanie, którego menedżer już nie widzi; pauzy (`interrupt`) nie są przerwanymi turami i działają jak dotąd.
- `start_message`: `before` liczony **przed** zamknięciem, więc `TurnResult.replies` nowej tury = `[INTERRUPTED_TURN_REPLY, <odpowiedź>]`; UI pokazuje obie wiadomości bez zmian w kodzie renderowania.
- `start_resume` bez pauzy i z przerwaną turą: zamknięcie, potem `NotPausedError("Poprzednia tura została przerwana; rozmowa nie czeka na odpowiedź.")` (409 `not_paused`). UI: nowy tekst w `render_error` dla `not_paused`: „Ta decyzja jest nieaktualna: rozmowa nie czeka już na odpowiedź.” i usunięcie przycisków pauzy (`_clear_pause_actions`).
- `get_thread`, `get_trace`: bez zapisu (odczyt nie zmienia stanu).
- Stan wniosku (`request`, `approvals`) zostaje: następna wiadomość może go zaktualizować albo przesłać ponownie jak w każdej innej turze.
- 409 `thread_stuck` zostaje w kontrakcie API jako błąd awaryjny (gdy zamknięcie się nie uda).
- Spike w A3: sprawdzić na `InMemorySaver`, że `update_state(..., as_node="reply_other")` czyści `next` dla przerwań w `understand`, `answer_documents`, `update_request`, `decide` i `approve` (przed `interrupt`); gdyby dla któregoś nie czyściło, wariant zapasowy: `as_node` = przerwany węzeł z jego krawędzią do END (dla `answer_documents`, `answer_policy`, `reply_other`) i `reply_other` dla reszty. Wynik spike'u (T1, 2026-11-07): `as_node="reply_other"` czyści `next` we wszystkich pięciu przypadkach, więc wariant zapasowy jest niepotrzebny (§14, P10).

### 8.2 Modele (`eval/dashboard/models.py`)

class MetricRow(BaseModel):
    name: str                      # po polsku, jak w raporcie Markdown etapu

class RunCalls(BaseModel):

    source: Literal["report", "dashboard.yaml"]
class StageCost(BaseModel):
    tokens_in: int | None; tokens_out: int | None
    ours_cents: float | None; reference_cents: float | None; course_cents: float | None
    includes_cached: bool          # True dla E5 (R1-3)
class StageSummary(BaseModel):
    stage: str; title: str; date: date; model: str | None; embedding_model: str | None
    report_md: str; related_md: list[str]; headline: str
    metrics: list[MetricRow]; calls: RunCalls | None; cost: StageCost | None; notes: list[str]
class MonthlyCost(BaseModel):      # scenariusz kursu: 400 pytań/dzień, 21 dni (E1)

class Dashboard(BaseModel):
